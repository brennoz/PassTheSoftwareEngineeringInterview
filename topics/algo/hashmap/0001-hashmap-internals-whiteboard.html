JAVA HashMap INTERNALS

// Java source (simplified)
class HashMap<K,V> {
    Node<K,V>[] table;          // array of buckets
    int size;                   // number of key-value pairs
    float loadFactor = 0.75f;   // resize when size > capacity * 0.75

hash() — why it XORs with high bits
// Java's actual spread function:
static final int hash(Object key) {
    int h = key.hashCode();
    return (h == 0) ? 0 : (h ^ (h >>> 16));
}
// Mixes high 16 bits into low 16 bits so small tables
// (capacity 16–32) don't ignore half the hash bits.
// bucket = hash & (capacity - 1)  ← bit-mask, not modulo

Frame 4 — hashCode() / equals() contract

hashCode() / equals() CONTRACT

The rule (Java spec):
If a.equals(b) → a.hashCode() == b.hashCode()
Converse is NOT required: same hash ≠ equal (that's just a collision)
// What HashMap.get() actually does — two-step lookup:
Node<K,V> getNode(Object key) {
    int hash = hash(key);
    int index = hash & (table.length - 1);  // Step 1: find bucket
    Node<K,V> node = table[index];
    while (node != null) {
        // Step 2: walk chain, match by hash THEN equals()
        if (node.hash == hash && node.key.equals(key))
            return node;
        node = node.next;
    }
    return null;
}
Bug: override equals(), forget hashCode()
class Order {
    String id;
    // @Override equals() ✓
    // hashCode() NOT overridden ✗
}
Order o1 = new Order("X");
Order o2 = new Order("X");
// o1.equals(o2) == true
// but different hashCode → different buckets
map.put(o1, "data");
map.get(o2); // → null ! (bucket miss)
Fix: always override both
// Java record auto-generates both:
record Order(String id) {}

// Or manually with Objects.hash:
@Override public int hashCode() {
    return Objects.hash(id);
}
@Override public boolean equals(Object o) {
    return o instanceof Order x
        && Objects.equals(id, x.id);
}

struct Node {

/* Three ways to access fields — KNOW ALL THREE */

Node node_val;           /* value (on stack) */
node_val.key = "hello";   /* dot  → direct field access */

Node *node_ptr = &node_val;         /* pointer to the node */
node_ptr->key = "hello";            /* arrow → (*node_ptr).key shorthand */
(*node_ptr).key = "hello";          /* same thing, long form */

/* malloc: allocate N bytes on the heap, returns void* */
Node *n = (Node *) malloc(sizeof(Node));
if (!n) { /* always check NULL — malloc fails under OOM */ }
free(n);                 /* release heap memory when done */

Frame 6 — C HashMap: djb2 + put/get

C HASHMAP IMPLEMENTATION

/* Entry: one key-value pair in a chain */
typedef struct Entry {
    char *key;
    char *value;
    struct Entry *next;
} Entry;

/* HashMap: array of Entry* pointers */
typedef struct {
    Entry **buckets;  /* pointer-to-pointer */
    int capacity;
} HashMap;

/* djb2: fast string hash (Bernstein 1991) */
unsigned int djb2(const char *str) {
    unsigned int h = 5381;
    int c;
    /* h = h*33 + c per character */
    while ((c = *str++))
        h = ((h << 5) + h) + c;
    return h;
}

/* Create: calloc zeroes all bucket pointers */
HashMap* hm_create(int cap) {
    HashMap *m = malloc(sizeof(HashMap));
    m->buckets = calloc(cap, sizeof(Entry*));
    m->capacity = cap;
    return m;
}
/* put: insert or update */
void hm_put(HashMap *m, const char *k, const char *v) {
    int idx = djb2(k) % m->capacity;
    Entry *e = m->buckets[idx];
    /* walk chain: update if key exists */
    while (e) {
        if (strcmp(e->key, k) == 0) {
            free(e->value);
            e->value = strdup(v); /* heap copy */
            return;
        }
        e = e->next;
    }
    /* not found: prepend new entry */
    Entry *ne = malloc(sizeof(Entry));
    ne->key   = strdup(k);
    ne->value = strdup(v);
    ne->next  = m->buckets[idx]; /* head-insert */
    m->buckets[idx] = ne;
}

/* get: return value or NULL */
char* hm_get(HashMap *m, const char *k) {
    int idx = djb2(k) % m->capacity;
    Entry *e = m->buckets[idx];
    while (e) {
        if (strcmp(e->key, k) == 0)
            return e->value;
        e = e->next;
    }
    return NULL;
}
strdup() = malloc + strcpy — caller must free()
calloc(n, size) = malloc(n*size) + zero-fills
strcmp(a,b)==0 means equal (not a==b which compares pointers)

Frame 7 — Distributed KV Store

DISTRIBUTED KV STORE — CONSISTENT HASHING

Consistent hash ring (3 servers, vnodes)
key hashes → first clockwise node = owner (B1)
HashMap → Distributed
hash(k) % cap→consistent_hash → ring
array bucket→server / vnode
collision chain→N replica servers
resize / rehash→rebalance on change
N / W / R quorums
N = replicas (e.g. 3)
W = write ACKs needed
R = read responses needed
W + R > N → strong consistency
Real systems
Redis Cluster — 16384 slots, CRC16
DynamoDB — vnodes, R/W configurable
Cassandra — tunable per query
